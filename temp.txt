R1 Identification d'un diamant
    :Mineral(?m) ^ :aPourElement(?m, ?e) ^ :Carbone(?e) ^ :dureteMohs(?m, ?d) ^ swrlb:greaterThanOrEqual(?d, 10)  ->  :Diamant(?m)
R2 Pierre tres dure
    :Gemme(?g) ^ :dureteMohs(?g, ?d) ^ swrlb:greaterThanOrEqual(?d, 9)  ->  :PierreTresDure(?g)
R3 Pierre de collection
    :PierrePrecieuse(?p) ^ :masseCarats(?p, ?m) ^ swrlb:greaterThanOrEqual(?m, 2.0)  ->  :PierreDeCollection(?p)
R4 Propagation fossile vers roche
    :Fossile(?f) ^ :estPreserveDans(?f, ?r) ^ :estTrouve(?f, ?c)  ->  :estTrouve(?r, ?c)
R5 Propagation roche vers constituant
    :Roche(?r) ^ :aPourConstituant(?r, ?m) ^ :estTrouve(?r, ?c)  ->  :estTrouve(?m, ?c)
R6 Flux commercial entre pays
    :PierreNaturelle(?p) ^ :estExportePar(?p, ?x) ^ :estImportePar(?p, ?y)  ->  :exporteVers(?x, ?y)
R7 Pays exportateur
    :Pays(?x) ^ :exporteVers(?x, ?y)  ->  :PaysExportateur(?x)
R8 Roche carbonatee
    :Roche(?r) ^ :contientElement(?r, ?e) ^ :Carbone(?e)  ->  :RocheCarbonatee(?r)
